<script>
(function(){
  var links = Array.prototype.slice.call(document.querySelectorAll('#rail a'));
  var secs  = links.map(function(a){ return document.getElementById(a.dataset.s); });
  var fill  = document.getElementById('railfill');
  var track = document.getElementById('railtrack');

  links.forEach(function(a){
    a.addEventListener('click', function(e){
      var t = document.getElementById(a.dataset.s);
      if (!t) return;
      e.preventDefault();
      t.scrollIntoView({ behavior:'smooth', block:'start' });
      if (history.replaceState) history.replaceState(null, '', '#' + a.dataset.s);
    });
  });

  function progress(){
    if (!fill || !track) return;
    var h = document.documentElement.scrollHeight - window.innerHeight;
    var p = h > 0 ? Math.min(1, Math.max(0, window.scrollY / h)) : 0;
    fill.style.height = (p * track.offsetHeight) + 'px';
  }
  var queued = false;
  window.addEventListener('scroll', function(){
    if (queued) return;
    queued = true;
    requestAnimationFrame(function(){ progress(); queued = false; });
  }, { passive:true });
  window.addEventListener('resize', progress, { passive:true });
  progress();

  if (!('IntersectionObserver' in window)) return;
  var io = new IntersectionObserver(function(entries){
    entries.forEach(function(e){
      var i = secs.indexOf(e.target);
      if (i < 0) return;
      links[i].classList.toggle('on', e.isIntersecting && e.intersectionRatio > 0.12);
    });
  }, { threshold:[0, .12, .5] });
  secs.forEach(function(s){ if (s) io.observe(s); });
})();
</script>
